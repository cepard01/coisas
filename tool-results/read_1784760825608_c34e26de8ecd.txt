   151→function GardenTab() {
   152→  const { t } = useI18n();
   153→  const { toast } = useToast();
   154→  const g = t.panel.dashboard.garden;
   155→
   156→  const [tick, setTick] = useState(0);
   157→  useEffect(() => {
   158→    const interval = setInterval(() => setTick((v) => v + 1), 2000);
   159→    return () => clearInterval(interval);
   160→  }, []);
   161→
   162→  const liveSlots = useMemo(() => {
   163→    return g.slots.map((slot, i) => {
   164→      const increment = tick * (0.3 + i * 0.1);
   165→      const progress = Math.min(slot.progress + increment, 99);
   166→      const remaining = Math.max(0, 100 - progress);
   167→      const minutes = Math.max(1, Math.ceil(remaining * 0.2));
   168→      return { ...slot, liveProgress: progress, liveEta: `~${minutes}m` };
   169→    });
   170→  }, [g.slots, tick]);
   171→
   172→  const nextReady = useMemo(() => {
   173→    const times = liveSlots.map((s) => Math.max(0, 100 - s.liveProgress));
   174→    const min = Math.min(...times);
   175→    return `~${Math.ceil(min * 0.2)}m`;
   176→  }, [liveSlots]);
   177→
   178→  const [harvested, setHarvested] = useState(false);
   179→  const handleHarvestAll = () => {
   180→    setHarvested(true);
   181→    setTimeout(() => setHarvested(false), 2000);
   182→    toast("Harvested 1 Sunflower · +400 🪙 +20 XP", "success");
   183→  };
   184→
   185→  return (
   186→    <div>
   187→      <TabHeader
   188→        title={g.title}
   189→        subtitle={g.subtitle}
   190→        action={
   191→          <div className="flex items-center gap-2">
   192→            <div className="hidden sm:flex items-center gap-2 rounded-full border border-border bg-background px-3 py-1.5">
   193→              <SunIcon size={14} className="text-gold-deep" />
   194→              <span className="text-xs font-medium text-foreground">{g.weather}</span>
   195→              <span className="font-mono text-[10px] text-muted-foreground">×1.5</span>
   196→            </div>
   197→            <motion.button
   198→              whileHover={{ scale: 1.03 }}
   199→              whileTap={{ scale: 0.97 }}
   200→              onClick={handleHarvestAll}
   201→              className={cn(
   202→                "inline-flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-xs font-medium transition-colors",
   203→                harvested
   204→                  ? "bg-sage text-white"
   205→                  : "bg-foreground text-background hover:bg-foreground/90"
   206→              )}
   207→            >
   208→              {harvested ? (
   209→                <>
   210→                  <CheckIcon size={13} /> +400 🪙 +20 XP
   211→                </>
   212→              ) : (
   213→                <>🌻 {g.harvestNow}</>
   214→              )}
   215→            </motion.button>
   216→          </div>
   217→        }
   218→      />
   219→
   220→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5 mb-7">
   221→        <MiniStat icon={SproutIcon} label={g.slotsUsed} value="4 / 6" />
   222→        <MiniStat icon={DropletIcon} label={g.humidity} value="50%" />
   223→        <MiniStat icon={ClockIcon} label={g.nextReady} value={nextReady} />
   224→        <MiniStat icon={SparkIcon} label={g.mutationHint} value="✓" />
   225→      </div>
   226→
   227→      <p className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground mb-3">
   228→        {g.slotsLabel}
   229→      </p>
   230→      <div className="grid grid-cols-2 sm:grid-cols-3 gap-2.5">
   231→        {liveSlots.map((slot, i) => (
   232→          <DashSlot
   233→            key={i}
   234→            emoji={slot.emoji}
   235→            name={slot.name}
   236→            progress={slot.liveProgress}
   237→            eta={slot.liveEta}
   238→          />
   239→        ))}
   240→        <DashSlot emoji={g.readySlot.emoji} name={g.readySlot.name} progress={100} ready />
   241→        <DashSlot empty />
   242→        <DashSlot empty />
   243→      </div>
   244→
   245→      <p className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground mt-7 mb-3">
   246→        {g.activityLabel}
   247→      </p>
   248→      <div className="space-y-2.5">
   249→        {g.activity.map((a, i) => (
   250→          <ActivityRow key={i} time={a.time} text={a.text} tone={a.tone} />
   251→        ))}
   252→      </div>
   253→    </div>
   254→  );

   320→        </>
   321→      )}
   322→    </motion.div>

   339→function WalletTab() {
   340→  const { t } = useI18n();
   341→  const w = t.panel.dashboard.wallet;
   342→
   343→  const [balance, setBalance] = useState(0);
   344→  useEffect(() => {
   345→    const target = 1240;
   346→    const duration = 1200;
   347→    const start = performance.now();
   348→    let raf: number;
   349→    const tick = (now: number) => {
   350→      const elapsed = now - start;
   351→      const progress = Math.min(elapsed / duration, 1);
   352→      const eased = 1 - Math.pow(1 - progress, 3);
   353→      setBalance(Math.round(target * eased));
   354→      if (progress < 1) raf = requestAnimationFrame(tick);
   355→    };
   356→    raf = requestAnimationFrame(tick);
   357→    return () => cancelAnimationFrame(raf);
   358→  }, []);
   359→
   360→  return (
   361→    <div>
   362→      <TabHeader title={w.title} subtitle={w.subtitle} />
   363→
   364→      <div className="border-b border-border pb-6 mb-6">

   368→            {balance.toLocaleString()}
   369→          </p>

   414→function Sparkline() {
   415→  const data = [820, 950, 780, 1100, 980, 1150, 1240];
   416→  const max = Math.max(...data);
   417→  const min = Math.min(...data);
   418→  const range = max - min || 1;
   419→  const points = data.map((v, i) => {
   420→    const x = (i / (data.length - 1)) * 280;
   421→    const y = 40 - ((v - min) / range) * 35;
   422→    return `${x},${y}`;
   423→  }).join(" ");
   424→
   425→  return (
   426→    <svg viewBox="0 0 280 50" className="w-full h-12" preserveAspectRatio="none">
   427→      <defs>
   428→        <linearGradient id="spark-grad" x1="0" y1="0" x2="0" y2="1">
   429→          <stop offset="0%" stopColor="oklch(0.46 0.06 145)" stopOpacity="0.2" />
   430→          <stop offset="100%" stopColor="oklch(0.46 0.06 145)" stopOpacity="0" />
   431→        </linearGradient>
   432→      </defs>
   433→      <motion.polyline
   434→        points={`0,50 ${points} 280,50`}
   435→        fill="url(#spark-grad)"
   436→        initial={{ opacity: 0 }}
   437→        animate={{ opacity: 1 }}
   438→        transition={{ duration: 0.5, delay: 0.5 }}
   439→      />
   440→      <motion.polyline
   441→        points={points}
   442→        fill="none"
   443→        stroke="oklch(0.46 0.06 145)"
   444→        strokeWidth="2"
   445→        strokeLinecap="round"
   446→        strokeLinejoin="round"
   447→        initial={{ pathLength: 0 }}
   448→        animate={{ pathLength: 1 }}
   449→        transition={{ duration: 1, delay: 0.3, ease: [0.16, 1, 0.3, 1] }}
   450→      />
   451→      {data.map((v, i) => {
   452→        const x = (i / (data.length - 1)) * 280;
   453→        const y = 40 - ((v - min) / range) * 35;
   454→        return (
   455→          <motion.circle
   456→            key={i}
   457→            cx={x}
   458→            cy={y}
   459→            r="2"
   460→            fill="oklch(0.46 0.06 145)"
   461→            initial={{ scale: 0 }}
   462→            animate={{ scale: 1 }}
   463→            transition={{ duration: 0.3, delay: 0.8 + i * 0.05 }}
   464→          />
   465→        );
   466→      })}
   467→    </svg>
   468→  );
   469→}

   594→/* ── Shop Tab (NEW) ──────────────────────────────────────────────────── */
   595→
   596→function ShopTab() {
   597→  const { t } = useI18n();
   598→  const { toast } = useToast();
   599→  const s = t.panel.dashboard.shop;
   600→  const [filter, setFilter] = useState<string>("all");
   601→  const [bought, setBought] = useState<string | null>(null);
   602→
   603→  const categories = [
   604→    { id: "all", label: "All" },
   605→    { id: "seeds", label: s.categories.seeds },
   606→    { id: "tools", label: s.categories.tools },
   607→    { id: "consumables", label: s.categories.consumables },
   608→    { id: "decor", label: s.categories.decor },
   609→  ];
   610→
   611→  const filtered = filter === "all" ? s.items : s.items.filter((i) => i.category === filter);
   612→
   613→  const handleBuy = (name: string, price: number) => {
   614→    setBought(name);
   615→    setTimeout(() => setBought(null), 1500);
   616→    toast(`Bought ${name} for ${price} Daisies`, "success");
   617→  };
   618→
   619→  return (
   620→    <div>
   621→      <TabHeader
   622→        title={s.title}
   623→        subtitle={s.subtitle}
   624→        action={
   625→          <div className="flex items-center gap-1.5 rounded-full border border-border bg-background px-3 py-1.5">
   626→            <span className="text-sm">🪙</span>
   627→            <span className="font-mono text-xs font-medium text-foreground tabular">1,240</span>
   628→          </div>
   629→        }
   630→      />
   631→
   632→      {/* Category filter */}
   633→      <div className="flex flex-wrap gap-1.5 mb-5">
   634→        {categories.map((cat) => (
   635→          <button
   636→            key={cat.id}
   637→            onClick={() => setFilter(cat.id)}
   638→            className={cn(
   639→              "rounded-full border px-3 py-1 text-xs font-medium transition-all",
   640→              filter === cat.id
   641→                ? "border-foreground bg-foreground text-background"
   642→                : "border-border bg-card text-muted-foreground hover:text-foreground hover:border-foreground/30"
   643→            )}
   644→          >
   645→            {cat.label}
   646→          </button>
   647→        ))}
   648→      </div>
   649→
   650→      {/* Shop items */}
   651→      <div className="grid sm:grid-cols-2 gap-3">
   652→        {filtered.map((item, i) => {
   653→          const canAfford = 1240 >= item.price;
   654→          const justBought = bought === item.name;
   655→          return (
   656→            <motion.div
   657→              key={i}
   658→              layout
   659→              initial={{ opacity: 0, scale: 0.95 }}
   660→              animate={{ opacity: 1, scale: 1 }}
   661→              transition={{ duration: 0.2, delay: Math.min(i * 0.03, 0.2) }}
   662→              className="card-hairline rounded-lg p-4 flex items-center gap-3"
   663→            >
   664→              <span className="grid place-items-center h-12 w-12 rounded-lg bg-secondary text-2xl shrink-0">
   665→                {item.emoji}
   666→              </span>
   667→              <div className="flex-1 min-w-0">
   668→                <div className="flex items-center gap-2">
   669→                  <p className="text-sm font-medium text-foreground truncate">{item.name}</p>
   670→                  {item.owned > 0 && (
   671→                    <span className="font-mono text-[10px] text-muted-foreground bg-secondary px-1.5 py-0.5 rounded">
   672→                      {s.owned}: {item.owned}
   673→                    </span>
   674→                  )}
   675→                </div>
   676→                <p className="text-[11px] text-muted-foreground text-pretty mt-0.5">{item.desc}</p>
   677→              </div>
   678→              <div className="shrink-0 text-right">
   679→                <p className="font-mono text-sm font-medium text-foreground tabular">🪙 {item.price}</p>
   680→                <motion.button
   681→                  whileHover={canAfford ? { scale: 1.05 } : {}}
   682→                  whileTap={canAfford ? { scale: 0.95 } : {}}
   683→                  onClick={() => canAfford && handleBuy(item.name, item.price)}
   684→                  disabled={!canAfford}
   685→                  className={cn(
   686→                    "mt-1 inline-flex items-center gap-1 rounded-md px-2 py-1 text-[11px] font-medium transition-colors",
   687→                    justBought
   688→                      ? "bg-sage text-white"
   689→                      : canAfford
   690→                        ? "bg-foreground text-background hover:bg-foreground/90"
   691→                        : "bg-secondary text-muted-foreground cursor-not-allowed"
   692→                  )}
   693→                >
   694→                  {justBought ? <><CheckIcon size={11} /> +1</> : canAfford ? s.buy : s.notEnough}
   695→                </motion.button>
   696→              </div>
   697→            </motion.div>
   698→          );
   699→        })}
   700→      </div>
   701→    </div>
   702→  );
   703→}
